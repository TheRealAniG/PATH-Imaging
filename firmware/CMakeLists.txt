cmake_minimum_required(VERSION 3.16)
project(path_camera LANGUAGES CXX)

option(PATH_CAMERA_TESTS_ONLY "Build with the test SDK double, without hardware" OFF)
add_executable(path-camera src/main.cpp)
target_compile_features(path-camera PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(path-camera PRIVATE /W4)
else()
  target_compile_options(path-camera PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PATH_CAMERA_TESTS_ONLY)
  target_include_directories(path-camera PRIVATE tests/fake_sdk)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME capture-behavior COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_capture.py $<TARGET_FILE:path-camera>)
else()
  find_package(vizionsdk REQUIRED)
  target_link_libraries(path-camera PRIVATE vizionsdk::VizionSDK)
  install(TARGETS path-camera RUNTIME DESTINATION bin)
endif()
